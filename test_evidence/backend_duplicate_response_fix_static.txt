            List<String> existingLines = jdbcTemplate.query(
                    "SELECT text FROM " + tableName + " ORDER BY line_no",
                    (rs, rowNum) -> rs.getString("text")
            );
            if (existingLines.equals(lines)) {
                return true;
            }
        }
        return false;
    }
